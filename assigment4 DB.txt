// 1. Insert 3 students with name, email, points, courses
db.students.insertMany([
  {
    name: "Ayan Omar",
    email: "ayan@gmail.com",
    points: 10,
    courses: ["html", "css"]
  },
  {
    name: "Sara Ali",
    email: "sara@gmail.com",
    points: 15,
    courses: ["javascript"]
  },
  {
    name: "Khalid Nur",
    email: "khalid@gmail.com",
    points: 5,
    courses: ["mongodb", "nodejs"]
  }
])

// 2. Use $set to update one email
db.students.updateOne(
  { name: "Ayan Omar" },
  { $set: { email: "ayan.new@gmail.com" } }
)

// 3. Use $inc to increase points
db.students.updateOne(
  { name: "Sara Ali" },
  { $inc: { points: 5 } }
)

// 4. Use $push to add a new course
db.students.updateOne(
  { name: "Khalid Nur" },
  { $push: { courses: "react" } }
)

// 5. Use $pull to remove a course
db.students.updateOne(
  { name: "Khalid Nur" },
  { $pull: { courses: "nodejs" } }
)

// 6. BONUS: update one student with $set, $inc, and $push together
db.students.updateOne(
  { name: "Sara Ali" },
  {
    $set: { email: "sara.updated@gmail.com" },
    $inc: { points: 10 },
    $push: { courses: "react" }
  }
)
